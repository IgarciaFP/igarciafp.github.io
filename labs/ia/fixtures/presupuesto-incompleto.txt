"use strict";

function calcularPresupuesto() {

    // Primero cogemos lo que hemos puesto en los campos
    let alojamiento = document.getElementById("alojamiento").value;
    let comida = document.getElementById("comida").value;
    let ocio = document.getElementById("ocio").value;

    let resultado = document.getElementById("resultado");

    // Miramos que no hayamos dejado ninguno vacío
    if (alojamiento == "" || comida == "" || ocio == "") {
        resultado.textContent = "No puedes dejar campos vacíos";
        return;
    }

    // Ahora pasamos los valores a números para poder sumarlos
    alojamiento = Number(alojamiento);
    comida = Number(comida);
    ocio = Number(ocio);

    // Comprobamos que sean números válidos
    if (!Number.isFinite(alojamiento) || !Number.isFinite(comida) || !Number.isFinite(ocio)) {
        resultado.textContent = "Introduce números válidos";
        return;
    }
